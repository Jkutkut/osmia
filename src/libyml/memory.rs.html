9/// Allocate memory using the system's `malloc` function.
10///
11/// This function allocates `size` bytes of uninitialized memory and returns a pointer to it.
12///
13/// # Arguments
14///
15/// * `size` - The number of bytes to allocate.
16///
17/// # Returns
18///
19/// Returns a pointer to the allocated memory, or a null pointer if the allocation failed.
20///
21/// # Safety
22///
23/// This function is unsafe because:
24/// - It directly calls the system's `malloc` function, which is not memory-safe.
25/// - The caller must ensure that the allocated memory is properly freed using `yaml_free`.
26/// - The caller is responsible for initializing the allocated memory before use.
27///
28/// # Examples
29///
30/// ```
31/// use libyml::memory::yaml_malloc;
32/// use libyml::yaml::size_t;
33/// use libyml::memory::yaml_free;
34///
35/// unsafe {
36///     let size: size_t = 1024;
37///     let ptr = yaml_malloc(size);
38///     if !ptr.is_null() {
39///         // Use the allocated memory
40///         // ...
41///         yaml_free(ptr);
42///     }
43/// }
44/// ```
45pub unsafe fn yaml_malloc(size: size_t) -> *mut c_void {
46    malloc(size)
47}

49/// Reallocate memory using the system's `realloc` function.
50///
51/// This function changes the size of the memory block pointed to by `ptr` to `size` bytes.
52///
53/// # Arguments
54///
55/// * `ptr` - A pointer to the memory block to reallocate. If null, this function behaves like `yaml_malloc`.
56/// * `size` - The new size of the memory block in bytes.
57///
58/// # Returns
59///
60/// Returns a pointer to the reallocated memory, which may be different from `ptr`, or a null pointer if the reallocation failed.
61///
62/// # Safety
63///
64/// This function is unsafe because:
65/// - It directly calls the system's `realloc` function, which is not memory-safe.
66/// - The caller must ensure that `ptr` is either null or was previously allocated by `yaml_malloc` or `yaml_realloc`.
67/// - The caller must ensure that the reallocated memory is properly freed using `yaml_free`.
68/// - The contents of the reallocated memory beyond the original size are undefined.
69///
70/// # Note
71///
72/// If the reallocation fails, the original memory block is left untouched and a null pointer is returned.
73///
74/// # Examples
75///
76/// ```
77/// use libyml::memory::{yaml_malloc, yaml_realloc, yaml_free};
78/// use libyml::yaml::size_t;
79///
80/// unsafe {
81///     let mut size: size_t = 1024;
82///     let mut ptr = yaml_malloc(size);
83///     if !ptr.is_null() {
84///         // Use the allocated memory
85///         // ...
86///         size = 2048;
87///         ptr = yaml_realloc(ptr, size);
88///         if !ptr.is_null() {
89///             // Use the reallocated memory
90///             // ...
91///             yaml_free(ptr);
92///         }
93///     }
94/// }
95/// ```
96pub unsafe fn yaml_realloc(
97    ptr: *mut c_void,
98    size: size_t,
99) -> *mut c_void {
100    if !ptr.is_null() {
101        realloc(ptr, size)
102    } else {
103        malloc(size)
104    }
105}

107/// Free memory allocated by `yaml_malloc` or `yaml_realloc`.
108///
109/// This function deallocates the memory previously allocated by `yaml_malloc` or `yaml_realloc`.
110///
111/// # Arguments
112///
113/// * `ptr` - A pointer to the memory block to free. If null, no operation is performed.
114///
115/// # Safety
116///
117/// This function is unsafe because:
118/// - It directly calls the system's `free` function, which is not memory-safe.
119/// - The caller must ensure that `ptr` was allocated by `yaml_malloc` or `yaml_realloc`.
120/// - After calling this function, `ptr` becomes invalid and must not be used.
121///
122/// # Examples
123///
124/// ```
125/// use libyml::memory::{yaml_malloc, yaml_free};
126/// use libyml::yaml::size_t;
127///
128/// unsafe {
129///     let size: size_t = 1024;
130///     let ptr = yaml_malloc(size);
131///     if !ptr.is_null() {
132///         // Use the allocated memory
133///         // ...
134///         yaml_free(ptr);
135///         // ptr is now invalid and must not be used
136///     }
137/// }
138/// ```
139pub unsafe fn yaml_free(ptr: *mut c_void) {
140    if !ptr.is_null() {
141        free(ptr);
142    }
143}

145/// Duplicate a string using the system's `malloc` function and manual copy due to type mismatch.
146///
147/// This function creates a new copy of the input string, allocating new memory for it.
148///
149/// # Arguments
150///
151/// * `str` - A pointer to the null-terminated string to duplicate.
152///
153/// # Returns
154///
155/// Returns a pointer to the newly allocated string, or a null pointer if the allocation failed or the input was null.
156///
157/// # Safety
158///
159/// This function is unsafe because:
160/// - It involves memory allocation and raw pointer manipulation.
161/// - The caller must ensure that `str` is a valid, null-terminated string.
162/// - The caller is responsible for freeing the returned pointer using `yaml_free`.
163///
164/// # Examples
165///
166/// ```
167/// use libyml::memory::{yaml_strdup, yaml_free};
168/// use libyml::yaml::yaml_char_t;
169/// use core::ffi::c_void;
170///
171/// unsafe {
172///     // Note: The cast to *const yaml_char_t is necessary because yaml_char_t
173///     // might not be the same as u8 on all systems.
174///     let original: *const yaml_char_t = b"Hello, world!\0".as_ptr() as *const yaml_char_t;
175///     let copy = yaml_strdup(original);
176///     if !copy.is_null() {
177///         // Use the duplicated string
178///         // ...
179///         yaml_free(copy as *mut c_void);
180///     }
181/// }
182/// ```
183pub unsafe fn yaml_strdup(str: *const yaml_char_t) -> *mut yaml_char_t {
184    if str.is_null() {
185        return ptr::null_mut();
186    }
187    let len = strlen(str as *const libc::c_char) as usize;
188    let new_size = (len + 1) * size_of::<yaml_char_t>();
189    let new_str =
190        malloc(new_size.try_into().unwrap()) as *mut yaml_char_t;
191    if new_str.is_null() {
192        return ptr::null_mut();
193    }
194    ptr::copy_nonoverlapping(str, new_str, len + 1);
195    new_str
196}
